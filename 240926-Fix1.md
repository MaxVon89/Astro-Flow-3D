"""
Data preprocessing module for Astro-Flow-3D.
Handles JWST FITS data ingestion and normalization.
"""

import os
import numpy as np
from astropy.io import fits
from typing import Tuple, List, Optional

class DataPreprocessor:
    """Handles preprocessing of astronomical data."""
    
    def __init__(self, config_path: str = "configs/preprocessing_config.yaml"):
        """
        Initialize the preprocessor with configuration.
        
        Args:
            config_path (str): Path to preprocessing configuration file
        """
        self.config_path = config_path
        
    def load_fits_data(self, filepath: str) -> Tuple[np.ndarray, dict]:
        """
        Load FITS data from file.
        
        Args:
            filepath (str): Path to FITS file
            
        Returns:
            Tuple[np.ndarray, dict]: Data array and header information
        """
        try:
            with fits.open(filepath) as hdul:
                data = hdul[0].data
                header = hdul[0].header
            return data, header
        except Exception as e:
            raise ValueError(f"Failed to load FITS file {filepath}: {str(e)}")
    
    def normalize_data(self, data: np.ndarray) -> np.ndarray:
        """
        Normalize data to standard range.
        
        Args:
            data (np.ndarray): Input data
            
        Returns:
            np.ndarray: Normalized data
        """
        # Clip outliers and normalize to [0, 1]
        data = np.clip(data, np.percentile(data, 1), np.percentile(data, 99))
        data = (data - data.min()) / (data.max() - data.min())
        return data
    
    def preprocess_pipeline(self, input_files: List[str], output_dir: str) -> List[str]:
        """
        Complete preprocessing pipeline for a list of input files.
        
        Args:
            input_files (List[str]): List of input file paths
            output_dir (str): Directory to save preprocessed files
            
        Returns:
            List[str]: List of output file paths
        """
        output_files = []
        
        for input_file in input_files:
            try:
                # Load data
                data, header = self.load_fits_data(input_file)
                
                # Normalize data
                normalized_data = self.normalize_data(data)
                
                # Save preprocessed data
                filename = os.path.basename(input_file)
                output_path = os.path.join(output_dir, f"preprocessed_{filename}")
                
                # Save as FITS file with original header
                fits.writeto(output_path, normalized_data, header, overwrite=True)
                output_files.append(output_path)
                
            except Exception as e:
                print(f"Error processing {input_file}: {str(e)}")
                continue
                
        return output_files

# Example usage
if __name__ == "__main__":
    preprocessor = DataPreprocessor()
    
    # Example preprocessing workflow
    input_files = ["data/raw/example.fits"]
    output_dir = "data/preprocessed"
    
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
        
    output_files = preprocessor.preprocess_pipeline(input_files, output_dir)
    print(f"Preprocessing complete. Output files: {output_files}")

"""
Tile generation module for Astro-Flow-3D.
Creates spatial tiles from preprocessed astronomical data.
"""

import os
import numpy as np
from typing import Tuple, List
import matplotlib.pyplot as plt

class TileGenerator:
    """Generates spatial tiles from astronomical data."""
    
    def __init__(self, tile_size: Tuple[int, int] = (256, 256)):
        """
        Initialize tile generator.
        
        Args:
            tile_size (Tuple[int, int]): Size of each tile (height, width)
        """
        self.tile_size = tile_size
        
    def generate_tiles(self, data: np.ndarray) -> List[np.ndarray]:
        """
        Generate tiles from input data.
        
        Args:
            data (np.ndarray): Input data array
            
        Returns:
            List[np.ndarray]: List of tile arrays
        """
        height, width = data.shape
        tile_h, tile_w = self.tile_size
        
        tiles = []
        
        # Calculate number of complete tiles
        n_tiles_h = height // tile_h
        n_tiles_w = width // tile_w
        
        for i in range(n_tiles_h):
            for j in range(n_tiles_w):
                start_h = i * tile_h
                start_w = j * tile_w
                
                tile = data[start_h:start_h + tile_h, start_w:start_w + tile_w]
                tiles.append(tile)
                
        return tiles
    
    def save_tiles(self, tiles: List[np.ndarray], output_dir: str) -> List[str]:
        """
        Save generated tiles to files.
        
        Args:
            tiles (List[np.ndarray]): List of tile arrays
            output_dir (str): Directory to save tiles
            
        Returns:
            List[str]: List of saved file paths
        """
        if not os.path.exists(output_dir):
            os.makedirs(output_dir)
            
        file_paths = []
        
        for i, tile in enumerate(tiles):
            filename = f"tile_{i:04d}.npy"
            filepath = os.path.join(output_dir, filename)
            np.save(filepath, tile)
            file_paths.append(filepath)
            
        return file_paths

# Example usage
if __name__ == "__main__":
    # Example with sample data
    sample_data = np.random.rand(1024, 1024)
    generator = TileGenerator(tile_size=(256, 256))
    
    tiles = generator.generate_tiles(sample_data)
    file_paths = generator.save_tiles(tiles, "data/tiles")
    
    print(f"Generated {len(tiles)} tiles")

"""
Dataset builder module for Astro-Flow-3D.
Creates dataset manifests and validates data integrity.
"""

import os
import json
import numpy as np
from typing import Dict, List, Any
import hashlib

class DatasetBuilder:
    """Builds datasets from processed tiles with validation."""
    
    def __init__(self, manifest_path: str = "dataset_manifest.json"):
        """
        Initialize dataset builder.
        
        Args:
            manifest_path (str): Path to save dataset manifest
        """
        self.manifest_path = manifest_path
        self.manifest = {
            "version": "1.0",
            "data_files": [],
            "metadata": {}
        }
    
    def add_data_file(self, file_path: str, file_type: str, 
                     metadata: Dict[str, Any] = None) -> None:
        """
        Add a data file to the dataset manifest.
        
        Args:
            file_path (str): Path to the data file
            file_type (str): Type of data file
            metadata (Dict[str, Any]): Additional metadata
        """
        # Calculate hash for validation
        file_hash = self._calculate_file_hash(file_path)
        
        file_info = {
            "path": file_path,
            "type": file_type,
            "hash": file_hash,
            "size": os.path.getsize(file_path),
            "metadata": metadata or {}
        }
        
        self.manifest["data_files"].append(file_info)
    
    def _calculate_file_hash(self, file_path: str) -> str:
        """
        Calculate SHA256 hash of a file.
        
        Args:
            file_path (str): Path to file
            
        Returns:
            str: SHA256 hash as hexadecimal string
        """
        sha256_hash = hashlib.sha256()
        
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(4096), b""):
                sha256_hash.update(chunk)
                
        return sha256_hash.hexdigest()
    
    def validate_dataset(self) -> bool:
        """
        Validate dataset integrity.
        
        Returns:
            bool: True if validation passes, False otherwise
        """
        # Check all files exist and have correct hashes
        for file_info in self.manifest["data_files"]:
            file_path = file_info["path"]
            
            if not os.path.exists(file_path):
                print(f"Missing file: {file_path}")
                return False
                
            # Verify hash
            calculated_hash = self._calculate_file_hash(file_path)
            if calculated_hash != file_info["hash"]:
                print(f"Hash mismatch for file: {file_path}")
                return False
                
        return True
    
    def save_manifest(self) -> None:
        """
        Save dataset manifest to JSON file.
        """
        with open(self.manifest_path, 'w') as f:
            json.dump(self.manifest, f, indent=2)
            
        print(f"Dataset manifest saved to {self.manifest_path}")

# Example usage
if __name__ == "__main__":
    builder = DatasetBuilder()
    
    # Add some sample files (these would be actual tile files)
    sample_files = ["data/tiles/tile_0000.npy", "data/tiles/tile_0001.npy"]
    
    for file_path in sample_files:

                file_path, 
                "tile",
                {"resolution": "256x256"}
            )
    
    # Validate and save
    if builder.validate_dataset():
        builder.save_manifest()
        print("Dataset validation passed")
    else:
        print("Dataset validation failed")
